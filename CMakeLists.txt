cmake_minimum_required(VERSION 3.22)
project(Flux VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
option(FLUX_CORE_ONLY "Build the portable motion tests without Qt" OFF)
if(BUILD_TESTING)
    add_executable(flux-motion-tests tests/motion_test.cpp)
    target_include_directories(flux-motion-tests PRIVATE src)
    add_test(NAME motion COMMAND flux-motion-tests)
    add_executable(flux-resize-tests tests/resize_test.cpp)
    target_include_directories(flux-resize-tests PRIVATE src)
    add_test(NAME resize COMMAND flux-resize-tests)
endif()
if(FLUX_CORE_ONLY)
    return()
endif()
find_package(Qt6 6.5 REQUIRED COMPONENTS Core Gui Qml Quick QuickControls2)
qt_standard_project_setup(REQUIRES 6.5)
if(Qt6_VERSION VERSION_GREATER_EQUAL 6.8)
    qt_policy(SET QTP0004 NEW)
endif()
qt_add_executable(flux src/main.cpp)
add_library(flux-document STATIC src/Document.cpp src/Document.hpp src/Render.cpp src/Render.hpp src/MotionPose.hpp
    src/TextLayout.cpp src/TextLayout.hpp src/TextMorph.cpp src/TextMorph.hpp)
target_include_directories(flux-document PUBLIC src)
target_link_libraries(flux-document PUBLIC Qt6::Core Qt6::Gui Qt6::Qml Qt6::Quick)
target_link_libraries(flux PRIVATE flux-document Qt6::QuickControls2)
qt_add_qml_module(flux URI Flux VERSION 1.0 RESOURCE_PREFIX /qt/qml
    QML_FILES qml/Main.qml qml/Tool.qml qml/ValueField.qml qml/NodeVisual.qml qml/SlideView.qml
              qml/MotionStage.qml qml/EditCanvas.qml qml/Inspector.qml qml/Presentation.qml qml/Presenter.qml
)
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    qt_add_executable(flux-document-tests tests/document_test.cpp)
    target_link_libraries(flux-document-tests PRIVATE flux-document Qt6::Test)
    add_test(NAME document COMMAND flux-document-tests)
    set_tests_properties(document PROPERTIES ENVIRONMENT "FLUX_DATA_DIR=${CMAKE_CURRENT_BINARY_DIR}/testdata;QT_QPA_PLATFORM=offscreen;QSG_RHI_BACKEND=software")
    qt_add_executable(flux-resize-ui-tests tests/resize_ui_test.cpp)
    target_compile_definitions(flux-resize-ui-tests PRIVATE FLUX_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
    target_link_libraries(flux-resize-ui-tests PRIVATE flux-document Qt6::Test Qt6::QuickControls2)
    add_test(NAME resize-ui COMMAND flux-resize-ui-tests)
    set_tests_properties(resize-ui PROPERTIES TIMEOUT 60 ENVIRONMENT "FLUX_DATA_DIR=${CMAKE_CURRENT_BINARY_DIR}/resize-testdata;QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software;QT_FORCE_STDERR_LOGGING=1")
    qt_add_executable(flux-text-morph-tests tests/text_morph_test.cpp)
    target_link_libraries(flux-text-morph-tests PRIVATE flux-document Qt6::Test)
    add_test(NAME text-morph COMMAND flux-text-morph-tests)
    set_tests_properties(text-morph PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software;QT_FORCE_STDERR_LOGGING=1")
    qt_add_executable(flux-morph-ui-tests tests/morph_ui_test.cpp)
    target_compile_definitions(flux-morph-ui-tests PRIVATE FLUX_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
    target_link_libraries(flux-morph-ui-tests PRIVATE flux-document Qt6::Test Qt6::QuickControls2)
    add_test(NAME morph-ui COMMAND flux-morph-ui-tests)
    set_tests_properties(morph-ui PROPERTIES TIMEOUT 30 ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software;QT_FORCE_STDERR_LOGGING=1")
    add_test(NAME qml-smoke COMMAND flux --smoke-test)
    set_tests_properties(qml-smoke PROPERTIES PASS_REGULAR_EXPRESSION "Flux UI smoke passed" TIMEOUT 30)
    set_tests_properties(qml-smoke PROPERTIES ENVIRONMENT "FLUX_DATA_DIR=${CMAKE_CURRENT_BINARY_DIR}/testdata;QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software;QSG_RHI_BACKEND=software")
    set_property(TEST qml-smoke APPEND PROPERTY ENVIRONMENT "QT_FORCE_STDERR_LOGGING=1")
endif()
include(GNUInstallDirs)
install(TARGETS flux RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES packaging/io.fluxpresent.Flux.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
install(FILES assets/io.fluxpresent.Flux.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps)
install(FILES packaging/io.fluxpresent.Flux.xml DESTINATION ${CMAKE_INSTALL_DATADIR}/mime/packages)
install(FILES packaging/io.fluxpresent.Flux.metainfo.xml DESTINATION ${CMAKE_INSTALL_DATADIR}/metainfo)
